# Tamra M5 — OCR for scanned PDF pages Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A scanned Thai PDF becomes searchable and highlightable (spec §13, M5). PDF pages with no text layer, or with a text layer that is visibly broken, are read by OCR on the indexer's worker. Their text is indexed like any other page, and the viewer highlights a cited passage on the real page image from the character boxes OCR found.

**Architecture:**
- A new package `tamra.ocr` runs PP-OCRv5 mobile (one detector, a Thai recogniser on every line, a Chinese recogniser on weak lines) on the onnxruntime and numpy the app already ships. Its detection post-processing and line crop are a numpy port of RapidOCR's (Apache-2.0).
- The three model files (29.4 MB) are a new catalog entry, `pp-ocrv5-mobile` (role `ocr`), downloaded by the existing `ModelManager` into `models/ppocr-v5/`. Settings > AI model gets a "Text recognition (OCR)" row next to "Search model".
- `parse_pdf` reports the pages OCR should read: no text layer, a garbled Chromium/Skia Thai layer, or text that is not drawn. With the models installed, the indexer renders each such page at 300 dpi (PDFium lock held only while drawing), reads it, and indexes the OCR text as that page's unit, with `"ocr": true` in the chunk location. One page at a time: shutdown and project deletion never wait for a whole file.
- Schema v4 adds `ocr_pages(file_id, page, text, boxes)`, written in the same transaction as the file's chunks, and `files.needs_ocr`, so installing the models re-queues every file that waited for them.
- The viewer reads an OCR page's text and boxes from `ocr_pages`. `locate` and the highlight rectangles work as for pdfium text; the location label reads "p. 4 · OCR".

**Tech Stack:** M4's stack. No new dependency, Python or npm: onnxruntime, numpy and pypdfium2 are already there.

**Spec:** `docs/superpowers/specs/2026-10-02-tamra-design.md` §4, §5, §8, §9, §10, §12, §13 (Task 1 updates §4, §5, §8, §9, §10 and §12 to what M5 builds). Inputs that are fixed and not reopened:
- `.superpowers/sdd/2026-10-21-tamra-m5-ocr/controller-decisions.md` (decisions 1–8, quoted where a task uses them);
- `docs/spikes/2026-10-m5-ocr-spike.md` (engine choice, model pins, measurements);
- the verified reference port in `.superpowers/sdd/2026-10-21-tamra-m5-ocr/port/` (gitignored). Its `geometry.py`, `engine.py`, `ocr_eval.py` and `test_ocr_geometry.py` are byte-for-byte the code blocks of Tasks 2 and 3. They were run while this plan was written: the geometry matches OpenCV (`minAreaRect` + `boxPoints` ordering on 200 random boxes, 0 mismatches; `warpPerspective` cubic, mean pixel difference 0.0005), and on the spike's test set the pipeline reproduces RapidOCR exactly on the Chinese photocopy pages (CER 0.021, was 0.100) with Thai 300 dpi CER 0.023 and English word hit 0.94.

```python
"""Image geometry for OCR, in numpy only.

The detection post-processing and the line crop follow RapidOCR 3.10.0
(https://github.com/RapidAI/RapidOCR, Apache-2.0): ch_ppocr_det/utils.py (DBPostProcess,
get_mini_boxes, box_score_fast, unclip, order_points_clockwise) and
utils/process_img.py (get_rotate_crop_image). OpenCV, pyclipper and shapely calls are
replaced by the numpy equivalents below.
"""

import math

import numpy as np

Box = np.ndarray  # (4, 2) float64 corners: top-left, top-right, bottom-right, bottom-left


def resize_bilinear(img: np.ndarray, h: int, w: int) -> np.ndarray:
    """Bilinear resize of an HxW or HxWxC image to float32, like cv2.INTER_LINEAR."""
    H, W = img.shape[:2]
    if (H, W) == (h, w):
        return img.astype(np.float32)
    ys = np.clip((np.arange(h) + 0.5) * H / h - 0.5, 0, H - 1)
    xs = np.clip((np.arange(w) + 0.5) * W / w - 0.5, 0, W - 1)
    y0, x0 = np.floor(ys).astype(int), np.floor(xs).astype(int)
    y1, x1 = np.minimum(y0 + 1, H - 1), np.minimum(x0 + 1, W - 1)
    f = img.astype(np.float32)
    if f.ndim == 2:
        f = f[:, :, None]
    wy, wx = (ys - y0)[:, None, None], (xs - x0)[None, :, None]
    top = f[y0][:, x0] * (1 - wx) + f[y0][:, x1] * wx
    bottom = f[y1][:, x0] * (1 - wx) + f[y1][:, x1] * wx
    out = top * (1 - wy) + bottom * wy
    return out[:, :, 0] if img.ndim == 2 else out


def components(mask: np.ndarray) -> list[np.ndarray]:
    """4-connected components of a bool mask, each as an (n, 3) int array of row runs
    (y, x_start, x_end) with x_end exclusive. Row runs plus union-find: no per-pixel loop."""
    parent: list[int] = []

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    runs: list[tuple[int, int, int, int]] = []
    previous: list[tuple[int, int, int]] = []
    padded = np.zeros((mask.shape[0], mask.shape[1] + 2), dtype=np.int8)
    padded[:, 1:-1] = mask
    edges = np.diff(padded, axis=1)
    for y in range(mask.shape[0]):
        starts = np.flatnonzero(edges[y] == 1)
        if not len(starts):
            previous = []
            continue
        ends = np.flatnonzero(edges[y] == -1)
        current = []
        j = 0
        for s, e in zip(starts.tolist(), ends.tolist(), strict=True):
            rid = len(parent)
            parent.append(rid)
            while j < len(previous) and previous[j][1] <= s:
                j += 1
            k = j
            while k < len(previous) and previous[k][0] < e:
                a, b = find(previous[k][2]), find(rid)
                if a != b:
                    parent[b] = a
                k += 1
            current.append((s, e, rid))
            runs.append((y, s, e, rid))
        previous = current
    groups: dict[int, list[tuple[int, int, int]]] = {}
    for y, s, e, rid in runs:
        groups.setdefault(find(rid), []).append((y, s, e))
    return [np.asarray(g, dtype=np.int64) for g in groups.values()]


def run_points(runs: np.ndarray) -> np.ndarray:
    """The pixel centres that bound a component: both ends of every run, as (x, y)."""
    ys = runs[:, 0].astype(np.float64)
    return np.concatenate(
        [np.stack([runs[:, 1], ys], 1), np.stack([runs[:, 2] - 1, ys], 1)]
    ).astype(np.float64)


def convex_hull(points: np.ndarray) -> np.ndarray:
    """Counter-clockwise convex hull (Andrew's monotone chain) of (n, 2) points."""
    pts = np.unique(points, axis=0)
    if len(pts) <= 2:
        return pts

    def cross(o, a, b) -> float:
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower: list = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list = []
    for p in pts[::-1]:
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return np.asarray(lower[:-1] + upper[:-1])


def min_area_rect(points: np.ndarray) -> tuple[Box, float]:
    """cv2.minAreaRect + boxPoints + RapidOCR's get_mini_boxes ordering: the corners of the
    smallest rotated rectangle around the points, and its shorter side."""
    hull = convex_hull(points)
    if len(hull) == 1:
        return np.repeat(hull, 4, axis=0).astype(np.float64), 0.0
    best = None
    for i in range(len(hull)):
        edge = hull[(i + 1) % len(hull)] - hull[i]
        length = math.hypot(edge[0], edge[1])
        if length == 0:
            continue
        u = edge / length
        v = np.array([-u[1], u[0]])
        pu, pv = hull @ u, hull @ v
        area = (pu.max() - pu.min()) * (pv.max() - pv.min())
        if best is None or area < best[0]:
            best = (area, u, v, pu.min(), pu.max(), pv.min(), pv.max())
    _, u, v, u0, u1, v0, v1 = best
    corners = [a * u + b * v for a, b in ((u0, v0), (u1, v0), (u1, v1), (u0, v1))]
    pts = sorted(corners, key=lambda p: p[0])
    left = (pts[0], pts[1]) if pts[1][1] > pts[0][1] else (pts[1], pts[0])  # (top, bottom)
    right = (pts[2], pts[3]) if pts[3][1] > pts[2][1] else (pts[3], pts[2])
    box = np.array([left[0], right[0], right[1], left[1]], dtype=np.float64)
    return box, float(min(u1 - u0, v1 - v0))


def _inside(box: np.ndarray, xs: np.ndarray, ys: np.ndarray) -> np.ndarray:
    """Which (xs, ys) points lie in the convex quadrilateral `box`, its edges included."""
    signs = []
    for i in range(4):
        (ax, ay), (bx, by) = box[i], box[(i + 1) % 4]
        signs.append((bx - ax) * (ys - ay) - (by - ay) * (xs - ax))
    s = np.stack(signs)
    return np.all(s >= -1e-9, axis=0) | np.all(s <= 1e-9, axis=0)


def box_score(prob: np.ndarray, box: Box) -> float:
    """box_score_fast: the mean probability inside the box (corners truncated to integers)."""
    h, w = prob.shape
    xmin = int(np.clip(np.floor(box[:, 0].min()), 0, w - 1))
    xmax = int(np.clip(np.ceil(box[:, 0].max()), 0, w - 1))
    ymin = int(np.clip(np.floor(box[:, 1].min()), 0, h - 1))
    ymax = int(np.clip(np.ceil(box[:, 1].max()), 0, h - 1))
    quad = box.astype(np.int32).astype(np.float64)
    ys, xs = np.mgrid[ymin : ymax + 1, xmin : xmax + 1]
    inside = _inside(quad, xs.astype(np.float64), ys.astype(np.float64))
    if not inside.any():
        return 0.0
    return float(prob[ymin : ymax + 1, xmin : xmax + 1][inside].mean())


def unclip(box: Box, ratio: float) -> np.ndarray:
    """DB unclip: pyclipper's round-join offset of the box by area * ratio / perimeter.

    pyclipper works on integer coordinates: the corners are truncated and the result is
    rounded. Only the straight offset edges are returned (8 points); the rounded corners lie
    inside their minimum-area rectangle, which is all the caller takes from them.
    """
    u, v = box[1] - box[0], box[3] - box[0]
    w, h = float(np.hypot(*u)), float(np.hypot(*v))
    if w == 0 or h == 0:
        return box.copy()
    d = (w * h) * ratio / (2 * (w + h))
    pts = np.trunc(box)
    centre = pts.mean(axis=0)
    out = []
    for i in range(4):
        a, b = pts[i], pts[(i + 1) % 4]
        edge = (b - a) / max(float(np.hypot(*(b - a))), 1e-9)
        normal = np.array([edge[1], -edge[0]])
        if np.dot(normal, (a + b) / 2 - centre) < 0:
            normal = -normal
        out += [a + d * normal, b + d * normal]
    return np.rint(np.asarray(out))


def order_clockwise(box: np.ndarray) -> Box:
    """order_points_clockwise: top-left, top-right, bottom-right, bottom-left."""
    xs = box[np.argsort(box[:, 0], kind="stable")]
    left, right = xs[:2], xs[2:]
    tl, bl = left[np.argsort(left[:, 1], kind="stable")]
    tr, br = right[np.argsort(right[:, 1], kind="stable")]
    return np.array([tl, tr, br, bl], dtype=np.float64)


def sort_boxes(boxes: list[Box], y_gap: float = 10.0) -> list[Box]:
    """Reading order: by the top-left corner's y, then x within lines closer than y_gap."""
    if not boxes:
        return []
    order = sorted(range(len(boxes)), key=lambda i: boxes[i][0, 1])
    line, last, keyed = 0, None, []
    for i in order:
        y = boxes[i][0, 1]
        if last is not None and y - last >= y_gap:
            line += 1
        last = y
        keyed.append((line, boxes[i][0, 0], i))
    return [boxes[i] for _, _, i in sorted(keyed)]


def _homography(src: np.ndarray, dst: np.ndarray) -> np.ndarray:
    """The 3x3 perspective transform that maps the 4 src points to the 4 dst points."""
    a, b = [], []
    for (x, y), (u, v) in zip(src, dst, strict=True):
        a.append([x, y, 1, 0, 0, 0, -u * x, -u * y])
        b.append(u)
        a.append([0, 0, 0, x, y, 1, -v * x, -v * y])
        b.append(v)
    m = np.linalg.solve(np.asarray(a, np.float64), np.asarray(b, np.float64))
    return np.append(m, 1.0).reshape(3, 3)


def _cubic(t: np.ndarray) -> list[np.ndarray]:
    """OpenCV's bicubic weights (a = -0.75) for the taps at -1, 0, 1, 2 around a sample."""
    a = -0.75

    def near(x):
        return ((a + 2) * x - (a + 3)) * x * x + 1

    def far(x):
        return ((a * x - 5 * a) * x + 8 * a) * x - 4 * a

    return [far(t + 1), near(t), near(1 - t), far(2 - t)]


def crop_quad(img: np.ndarray, box: Box) -> tuple[np.ndarray, bool]:
    """get_rotate_crop_image: the box warped upright (bicubic, border replicate). A crop at
    least 1.5 times taller than wide is turned 90 degrees; the flag says so."""
    w = int(max(np.hypot(*(box[0] - box[1])), np.hypot(*(box[2] - box[3]))))
    h = int(max(np.hypot(*(box[0] - box[3])), np.hypot(*(box[1] - box[2]))))
    w, h = max(w, 1), max(h, 1)
    target = np.array([[0, 0], [w, 0], [w, h], [0, h]], np.float64)
    inverse = np.linalg.inv(_homography(box, target))
    gy, gx = np.mgrid[0:h, 0:w].astype(np.float64)
    q = inverse @ np.stack([gx.ravel(), gy.ravel(), np.ones(gx.size)])
    sx, sy = (q[0] / q[2]).reshape(h, w), (q[1] / q[2]).reshape(h, w)
    x0, y0 = np.floor(sx).astype(int), np.floor(sy).astype(int)
    wx, wy = _cubic(sx - x0), _cubic(sy - y0)
    f = img.astype(np.float32)
    if f.ndim == 2:
        f = f[:, :, None]
    H, W = f.shape[:2]
    out = np.zeros((h, w, f.shape[2]), np.float32)
    for j in range(4):
        yy = np.clip(y0 + j - 1, 0, H - 1)
        row = np.zeros_like(out)
        for i in range(4):
            xx = np.clip(x0 + i - 1, 0, W - 1)
            row += f[yy, xx] * wx[i][:, :, None]
        out += row * wy[j][:, :, None]
    crop = np.clip(np.rint(out), 0, 255).astype(np.uint8)
    if img.ndim == 2:
        crop = crop[:, :, 0]
    if h / w >= 1.5:
        return np.rot90(crop), True
    return crop, False


def slice_box(box: Box, a: float, b: float) -> tuple[float, float, float, float]:
    """Axis-aligned bounds (x0, y0, x1, y1) of the part [a, b) of the box's width."""
    top0, top1 = box[0] + (box[1] - box[0]) * a, box[0] + (box[1] - box[0]) * b
    bottom0, bottom1 = box[3] + (box[2] - box[3]) * a, box[3] + (box[2] - box[3]) * b
    pts = np.array([top0, top1, bottom0, bottom1])
    return (
        float(pts[:, 0].min()),
        float(pts[:, 1].min()),
        float(pts[:, 0].max()),
        float(pts[:, 1].max()),
    )
```

Notes for the implementer (why the code is shaped this way; keep it):
- `components` + `run_points` stand in for `cv2.findContours`: the convex hull of the run end points is the hull of the contour, so `min_area_rect` gets the same rectangle (`RETR_LIST` hole contours are skipped; they score below the box threshold anyway).
- `unclip` is not the exact geometric offset on purpose. pyclipper truncates the box to integers and rounds its output, and those ±1 px decide whether a Chinese character survives on a noisy page: the exact-float version measured CER 0.042 on the photocopy set, this one 0.021 (RapidOCR's own number).

```python
"""Scoring and scan simulation for the OCR quality gate (the M5 spike's method, numpy only)."""

import re
import unicodedata
from collections import Counter

import numpy as np


def normalise(text: str, lang: str) -> str:
    """NFKC; English keeps single spaces, Thai and Chinese drop all whitespace."""
    text = unicodedata.normalize("NFKC", text)
    return re.sub(r"\s+", " ", text).strip() if lang == "en" else re.sub(r"\s+", "", text)


def levenshtein(a: str, b: str) -> int:
    previous = list(range(len(b) + 1))
    for i, ca in enumerate(a, 1):
        current = [i]
        for j, cb in enumerate(b, 1):
            current.append(min(previous[j] + 1, current[j - 1] + 1, previous[j - 1] + (ca != cb)))
        previous = current
    return previous[-1]


def cer(truth: str, ocr: str, lang: str) -> float:
    t, o = normalise(truth, lang), normalise(ocr, lang)
    return levenshtein(t, o) / max(len(t), 1)


def word_hit(truth: str, ocr: str) -> float:
    """The share of truth words (lowercased, punctuation stripped, as a multiset) in the OCR."""

    def words(s: str) -> Counter:
        found = (re.sub(r"[^\w]", "", w.lower()) for w in unicodedata.normalize("NFKC", s).split())
        return Counter(w for w in found if w)

    t, o = words(truth), words(ocr)
    return sum((t & o).values()) / max(sum(t.values()), 1)


def photocopy(gray: np.ndarray, seed: int, degrees: float = 1.0, noise: float = 18.0) -> np.ndarray:
    """A grayscale page with Gaussian noise, turned `degrees` counter-clockwise on white
    (canvas expanded, bilinear), as the spike made its photocopy variant with PIL."""
    rng = np.random.default_rng(seed)
    noisy = np.clip(gray.astype(np.float32) + rng.normal(0, noise, gray.shape), 0, 255)
    h, w = noisy.shape
    t = np.radians(degrees)
    c, s = np.cos(t), np.sin(t)
    out_w = int(np.ceil(abs(w * c) + abs(h * s)))
    out_h = int(np.ceil(abs(w * s) + abs(h * c)))
    ys, xs = np.mgrid[0:out_h, 0:out_w].astype(np.float32)
    xs, ys = xs - out_w / 2, ys - out_h / 2
    sx = c * xs - s * ys + w / 2  # inverse rotation of each output pixel into the source
    sy = s * xs + c * ys + h / 2
    x0, y0 = np.floor(sx).astype(int), np.floor(sy).astype(int)
    fx, fy = sx - x0, sy - y0
    padded = np.pad(noisy, 1, constant_values=255)

    def at(y, x):
        return padded[np.clip(y + 1, 0, h + 1), np.clip(x + 1, 0, w + 1)]

    out = (at(y0, x0) * (1 - fx) + at(y0, x0 + 1) * fx) * (1 - fy) + (
        at(y0 + 1, x0) * (1 - fx) + at(y0 + 1, x0 + 1) * fx
    ) * fy
    return np.clip(np.rint(out), 0, 255).astype(np.uint8)
```

```python
"""PP-OCRv5 mobile on onnxruntime: detection, a Thai-first recogniser cascade, character boxes."""

import math
from dataclasses import dataclass
from pathlib import Path

import numpy as np
import onnxruntime as ort
import pypdfium2 as pdfium

from tamra.ingest.parsers import PDFIUM_LOCK
from tamra.ocr.geometry import (
    box_score,
    components,
    crop_quad,
    min_area_rect,
    order_clockwise,
    resize_bilinear,
    run_points,
    slice_box,
    sort_boxes,
    unclip,
)

MODEL_DIR = "ppocr-v5"  # under the models folder
DET_FILE = "ch_PP-OCRv5_det_mobile.onnx"
REC_TH_FILE = "th_PP-OCRv5_rec_mobile.onnx"  # Thai + Latin: run on every line
REC_ZH_FILE = "ch_PP-OCRv5_rec_mobile.onnx"  # Chinese + Latin: only where the Thai one is weak
MODEL_FILES = (DET_FILE, REC_TH_FILE, REC_ZH_FILE)

OCR_DPI = 300  # page render resolution for OCR
OCR_THREADS = 4  # onnxruntime intra-op threads: leaves cores for the UI and the embedder
MAX_SIDE = 2000  # the page is scaled to at most this many pixels on its long side
DET_MIN_SIDE = 736
DET_THRESH, BOX_THRESH, UNCLIP_RATIO, MAX_CANDIDATES = 0.3, 0.5, 1.6, 1000
CASCADE = 0.9  # a line the Thai recogniser scores below this is also read as Chinese
MIN_SCORE = 0.5  # lines scoring lower are dropped
REC_HEIGHT, REC_MIN_WIDTH, REC_BATCH = 48, 320, 6


@dataclass(frozen=True)
class OcrPageResult:
    text: str  # the lines in reading order, joined with "\n"
    boxes: np.ndarray  # float32 (len(text), 4): x0, y0, x1, y1 as page fractions; "\n" is zeros


def render_page(path: Path, page: int, dpi: int = OCR_DPI) -> np.ndarray:
    """The 1-based page drawn at `dpi` as an (h, w, 3 or 4) BGR(x) array. PDFium is held only
    while drawing: recognition runs without the lock."""
    with PDFIUM_LOCK:
        pdf = pdfium.PdfDocument(path)
        try:
            pdf_page = pdf[page - 1]
            try:
                bitmap = pdf_page.render(scale=dpi / 72)
                try:
                    return bitmap.to_numpy().copy()  # a copy: the bitmap is freed below
                finally:
                    bitmap.close()
            finally:
                pdf_page.close()
        finally:
            pdf.close()


def _session(path: Path, threads: int) -> ort.InferenceSession:
    options = ort.SessionOptions()
    options.intra_op_num_threads = threads
    return ort.InferenceSession(str(path), options, providers=["CPUExecutionProvider"])


def _normalise(img: np.ndarray) -> np.ndarray:
    return ((img / 255.0 - 0.5) / 0.5).astype(np.float32)


class _Recogniser:
    def __init__(self, path: Path, threads: int):
        self.session = _session(path, threads)
        chars = self.session.get_modelmeta().custom_metadata_map["character"].splitlines()
        self.chars = ["", *chars, " "]  # CTC blank first; PP-OCR adds the space last

    def __call__(
        self, crops: list[np.ndarray]
    ) -> list[tuple[str, float, list[tuple[str, float, float]]]]:
        """(text, mean score, [(char, x0, x1) as fractions of the crop width]) per crop."""
        order = np.argsort([c.shape[1] / c.shape[0] for c in crops], kind="stable")
        out: list = [None] * len(crops)
        for start in range(0, len(crops), REC_BATCH):
            batch = order[start : start + REC_BATCH]
            ratio = max(
                REC_MIN_WIDTH / REC_HEIGHT, *(crops[i].shape[1] / crops[i].shape[0] for i in batch)
            )
            width = int(REC_HEIGHT * ratio)
            x = np.zeros((len(batch), 3, REC_HEIGHT, width), np.float32)
            widths = []
            for n, i in enumerate(batch):
                crop = crops[i]
                w = min(width, math.ceil(REC_HEIGHT * crop.shape[1] / crop.shape[0]))
                x[n, :, :, :w] = _normalise(resize_bilinear(crop, REC_HEIGHT, w)).transpose(2, 0, 1)
                widths.append(w)
            probs = self.session.run(None, {self.session.get_inputs()[0].name: x})[0]
            steps = probs.shape[1]
            for n, i in enumerate(batch):
                ids, best = probs[n].argmax(1), probs[n].max(1)
                text, scores, spans, last = [], [], [], 0
                for t in range(steps):
                    k = int(ids[t])
                    if k and k != last:
                        text.append(self.chars[k])
                        scores.append(float(best[t]))
                        x0 = t * width / steps / widths[n]
                        spans.append(
                            (
                                self.chars[k],
                                min(x0, 1.0),
                                min((t + 1) * width / steps / widths[n], 1.0),
                            )
                        )
                    last = k
                out[i] = ("".join(text), float(np.mean(scores)) if scores else 0.0, spans)
        return out


class OcrEngine:
    def __init__(self, det: ort.InferenceSession, thai: _Recogniser, chinese: _Recogniser):
        self._det, self._thai, self._chinese = det, thai, chinese

    @classmethod
    def load(cls, model_dir: Path, threads: int = OCR_THREADS) -> "OcrEngine":
        return cls(
            _session(model_dir / DET_FILE, threads),
            _Recogniser(model_dir / REC_TH_FILE, threads),
            _Recogniser(model_dir / REC_ZH_FILE, threads),
        )

    def read(self, image: np.ndarray) -> OcrPageResult:
        img = image[:, :, :3] if image.ndim == 3 else np.repeat(image[:, :, None], 3, axis=2)
        h, w = img.shape[:2]
        if max(h, w) > MAX_SIDE:  # RapidOCR's reduce_max_side: sides rounded to multiples of 32
            r = MAX_SIDE / max(h, w)
            rh, rw = round(int(h * r) / 32) * 32, round(int(w * r) / 32) * 32
            img = np.clip(np.rint(resize_bilinear(img, rh, rw)), 0, 255).astype(np.uint8)
            h, w = img.shape[:2]
        boxes = self._detect(img)
        crops = [crop_quad(img, b) for b in boxes]
        lines = self._recognise([c for c, _ in crops])
        text: list[str] = []
        cells: list[tuple[float, float, float, float]] = []
        for box, (_, turned), (line, score, spans) in zip(boxes, crops, lines, strict=True):
            if score < MIN_SCORE or not line.strip():
                continue
            if text:
                text.append("\n")
                cells.append((0.0, 0.0, 0.0, 0.0))
            for char, a, b in spans:
                x0, y0, x1, y1 = slice_box(box, 0.0, 1.0) if turned else slice_box(box, a, b)
                cell = (x0 / w, y0 / h, x1 / w, y1 / h)
                text.append(char)
                cells.extend([cell] * len(char))  # a dictionary entry may hold several code points
        joined = "".join(text)
        grid = np.clip(np.asarray(cells, np.float32).reshape(-1, 4), 0.0, 1.0)
        return OcrPageResult(joined, grid)

    def _detect(self, img: np.ndarray) -> list[np.ndarray]:
        h, w = img.shape[:2]
        r = DET_MIN_SIDE / min(h, w) if min(h, w) < DET_MIN_SIDE else 1.0
        dh, dw = max(32, round(int(h * r) / 32) * 32), max(32, round(int(w * r) / 32) * 32)
        x = _normalise(resize_bilinear(img, dh, dw)).transpose(2, 0, 1)[None]
        prob = self._det.run(None, {self._det.get_inputs()[0].name: x})[0][0, 0]
        mask = prob > DET_THRESH
        mask[:, 1:] |= mask[:, :-1]  # cv2.dilate with a 2x2 kernel
        mask[1:, :] |= mask[:-1, :]
        out = []
        for runs in components(mask)[:MAX_CANDIDATES]:
            box, short = min_area_rect(run_points(runs))
            if short < 3 or box_score(prob, box) < BOX_THRESH:
                continue
            box, short = min_area_rect(unclip(box, UNCLIP_RATIO))
            if short < 5:
                continue
            box[:, 0] = np.clip(np.round(box[:, 0] / dw * w), 0, w)
            box[:, 1] = np.clip(np.round(box[:, 1] / dh * h), 0, h)
            box = order_clockwise(box)
            box[:, 0] = np.clip(box[:, 0], 0, w - 1)
            box[:, 1] = np.clip(box[:, 1], 0, h - 1)
            if int(np.hypot(*(box[0] - box[1]))) <= 3 or int(np.hypot(*(box[0] - box[3]))) <= 3:
                continue
            out.append(box)
        return sort_boxes(out)

    def _recognise(self, crops: list[np.ndarray]) -> list[tuple[str, float, list]]:
        if not crops:
            return []
        lines = self._thai(crops)
        weak = [i for i, line in enumerate(lines) if line[1] < CASCADE]
        if weak:
            for i, line in zip(weak, self._chinese([crops[i] for i in weak]), strict=True):
                if line[1] > lines[i][1]:
                    lines[i] = line
        return lines
```

```python
def image_pdf(src: Path, dest: Path, dpi: int = 200) -> Path:
    """A scan of src: every page drawn as a grayscale image at `dpi`, with no text layer."""
    doc = pdfium.PdfDocument(src)
    out = pdfium.PdfDocument.new()
    try:
        for index in range(len(doc)):
            page = doc[index]
            width, height = page.get_size()
            gray = np.ascontiguousarray(page.render(scale=dpi / 72, grayscale=True).to_numpy())
            if gray.ndim == 3:
                gray = gray[:, :, 0]
            bitmap = pdfium.PdfBitmap.new_native(
                gray.shape[1], gray.shape[0], pdfium_c.FPDFBitmap_Gray
            )
            bitmap.to_numpy()[...] = gray
            target = out.new_page(width, height)
            image = pdfium.PdfImage.new(out)
            image.set_bitmap(bitmap)
            image.set_matrix(pdfium.PdfMatrix().scale(width, height))
            target.insert_obj(image)
            target.gen_content()
        dest.parent.mkdir(parents=True, exist_ok=True)
        out.save(dest)
    finally:
        out.close()
        doc.close()
    return dest
```

```python
def thai_echoes(text: str) -> int:
    """Signs of a Chromium/Skia-printed Thai text layer: a cluster repeated after its word
    ("บอร์ดร์"), a short line repeating the tail of the line above ("สรุป\\nรุ"), and marks with
    no base letter."""
    count = len(_ECHO.findall(text)) + len(_ORPHAN_MARK.findall(text))
    previous = ""
    for line in text.splitlines():
        line = line.strip()
        if line and len(line) <= 3 and "\u0e01" <= line[0] <= "\u0e2e" and line in previous[-4:]:
            count += 1
        if line:
            previous = line
    return count


def _not_drawn(page, text: str) -> bool:
    """A page with plenty of text whose render is almost blank: its text is invisible."""
    chars = len(text.strip())
    if chars < INVISIBLE_MIN_CHARS:
        return False
    bitmap = page.render(scale=INK_SCALE, grayscale=True)
    try:
        dark = int((bitmap.to_numpy() < 224).sum())
    finally:
        bitmap.close()
    return dark / chars < INVISIBLE_INK_PER_CHAR


def with_ocr(doc: ParsedDoc, pages: dict[int, str]) -> ParsedDoc:
    """doc with the text of these pages replaced by OCR text; an empty text drops the page."""
    units = [u for u in doc.units if u.page not in pages]
    units += [Unit(text=t, page=p, ocr=True) for p, t in pages.items() if t.strip()]
    return replace(doc, units=sorted(units, key=lambda u: u.page))
```

```python
from tamra.ocr import OcrPageResult


class FakeOcr:
    """Reads every page as the same text with one box per code point. `fail_calls` are the call
    numbers (1-based) that raise; `on_read(call)` runs at the start of each call."""

    BOX = (0.1, 0.2, 0.3, 0.25)

    def __init__(self, text="Scanned page about the boiler warranty.", fail_calls=(), on_read=None):
        self.text = text
        self.fail_calls = set(fail_calls)
        self.on_read = on_read
        self.calls = 0

    def read(self, image: np.ndarray) -> OcrPageResult:
        self.calls += 1
        if self.on_read is not None:
            self.on_read(self.calls)
        if self.calls in self.fail_calls:
            raise RuntimeError("OCR broke")
        boxes = np.tile(np.array(self.BOX, np.float32), (len(self.text), 1))
        return OcrPageResult(self.text, boxes)
```

```python
    def _read_pages(
        self, file_id: int, path: Path, pages: tuple[int, ...], engine: OcrLike
    ) -> tuple[dict[int, str], list[OcrPage], list[int]] | None:
        """OCR the pages one at a time: (page -> text, rows to store, pages that failed). None
        when indexing must stop between pages: on shutdown (the file goes back to pending) or
        when the file left the index (its project was deleted, or the file was removed)."""
        texts: dict[int, str] = {}
        rows: list[OcrPage] = []
        failed: list[int] = []
        try:
            for n, page in enumerate(pages, start=1):
                if self._stopping.is_set():
                    self._store.set_file_status(file_id, "pending")
                    return None
                if self._store.get_file(file_id) is None:
                    return None
                with self._lock:
                    self._reading = (n, len(pages))
                try:
                    result = engine.read(render_page(path, page))  # PDFium is held only to draw
                except Exception:  # one unreadable page must not lose the file
                    log.warning("OCR of page %s of %s failed", page, path.name, exc_info=True)
                    failed.append(page)
                    continue
                texts[page] = result.text
                if result.text:
                    boxes = np.asarray(result.boxes, dtype="<f4").tobytes()
                    rows.append(OcrPage(page, result.text, boxes))
        finally:
            with self._lock:
                self._reading = None
        return texts, rows, failed
```

```python
    def ocr_engine(self) -> OcrLike | None:
        """The OCR models, loaded on first use and kept. None while they are not installed or
        cannot be loaded; scanned pages then wait, and the next file tries again."""
        with self._ocr_lock:
            if self._ocr is None:
                try:
                    self._ocr = self._ocr_factory()
                except Exception as e:  # damaged or unreadable model files
                    log.warning("the OCR models cannot be loaded: %s", e)
            return self._ocr

    def _load_ocr(self) -> OcrLike | None:
        entry = self._catalog.ocr()
        if entry is None or entry.id not in self._catalog.installed(self._models_dir):
            return None
        return OcrEngine.load(self._models_dir / OCR_MODEL_DIR)

    def _model_installed(self, model_id: str) -> None:
        """The search model arrived: files that paused for lack of it are indexed now. The OCR
        models arrived: every file with pages left for OCR is read again, in every project."""
        ocr = self._catalog.ocr()
        if ocr is not None and model_id == ocr.id:
            self.store.requeue_ocr_files()
            self.indexer.request_reconcile()
        elif model_id == self._catalog.embedding().id:
            self.indexer.request_reconcile()
```

```python
def ocr_rects(
    text: str, boxes: bytes, start: int, end: int
) -> list[tuple[float, float, float, float]]:
    """Boxes covering OCR page text [start, end): the characters of each line run merged into one
    rectangle, as (x0, y0, x1, y1) page fractions with a top-left origin, like pdf_rects."""
    grid = np.frombuffer(boxes, dtype="<f4").reshape(-1, 4)
    start, end = max(start, 0), min(end, len(text), len(grid))
    rects = []
    run = start
    for i in range(start, end + 1):
        if i == end or text[i] == "\n":
            cells = grid[run:i]
            cells = cells[(cells[:, 2] > cells[:, 0]) & (cells[:, 3] > cells[:, 1])]
            if len(cells):
                rects.append(
                    (
                        float(cells[:, 0].min()),
                        float(cells[:, 1].min()),
                        float(cells[:, 2].max()),
                        float(cells[:, 3].max()),
                    )
                )
            run = i + 1
    return rects
```

```python
def link_models(src: Path, dest: Path) -> None:
    """Make the embedding and OCR model folders of src available under dest without copying
    (a copy when hard links fail). The LLM is not linked: it comes from the import step."""
    for folder in ("bge-m3", "ppocr-v5"):
        if not (src / folder).is_dir():

        for f in (src / folder).iterdir():
            target = dest / folder / f.name
            target.parent.mkdir(parents=True, exist_ok=True)
```
